## Step 2: 进程创建实验

```c
// fork_demo.c — 演示 fork 创建子进程及进程并发
#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>
#include <sys/types.h>

int main() {
    pid_t pid1, pid2;

    printf("父进程 PID = %d 开始\n", getpid());

    pid1 = fork();
    if (pid1 == 0) {
        // 子进程 1
        printf("  [子进程1] PID=%d, 父PID=%d\n", getpid(), getppid());
        sleep(1);
        printf("  [子进程1] 结束\n");
        return 0;
    }

    pid2 = fork();
    if (pid2 == 0) {
        // 子进程 2
        printf("  [子进程2] PID=%d, 父PID=%d\n", getpid(), getppid());
        sleep(2);
        printf("  [子进程2] 结束\n");
        return 0;
    }

    // 父进程等待两个子进程结束
    waitpid(pid1, NULL, 0);
    waitpid(pid2, NULL, 0);
    printf("父进程：所有子进程已结束\n");
    return 0;
}
```

```bash
gcc fork_demo.c -o fork_demo && ./fork_demo
```

## Step 3: 信号量互斥实验

```c
// semaphore_demo.c — 使用 POSIX 信号量实现生产者消费者
#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>
#include <unistd.h>

#define BUFFER_SIZE 5
int buffer[BUFFER_SIZE];
int in = 0, out = 0;

sem_t empty;         // 空闲缓冲区数
sem_t full;          // 产品数
pthread_mutex_t mutex;  // 互斥锁

void* producer(void* arg) {
    for (int i = 0; i < 10; i++) {
        sem_wait(&empty);           // P(empty) — 消耗一个空闲位置
        pthread_mutex_lock(&mutex); // P(mutex)

        buffer[in] = i;
        printf("生产: buffer[%d] = %d\n", in, i);
        in = (in + 1) % BUFFER_SIZE;

        pthread_mutex_unlock(&mutex); // V(mutex)
        sem_post(&full);              // V(full) — 增加一个产品
        usleep(100000);               // 模拟生产耗时
    }
    return NULL;
}

void* consumer(void* arg) {
    for (int i = 0; i < 10; i++) {
        sem_wait(&full);            // P(full) — 消耗一个产品
        pthread_mutex_lock(&mutex);

        int item = buffer[out];
        printf("消费: buffer[%d] = %d\n", out, item);
        out = (out + 1) % BUFFER_SIZE;

        pthread_mutex_unlock(&mutex);
        sem_post(&empty);            // V(empty)
        usleep(200000);              // 模拟消费耗时
    }
    return NULL;
}

int main() {
    pthread_t prod, cons;

    sem_init(&empty, 0, BUFFER_SIZE);   // empty = n
    sem_init(&full, 0, 0);              // full = 0
    pthread_mutex_init(&mutex, NULL);

    pthread_create(&prod, NULL, producer, NULL);
    pthread_create(&cons, NULL, consumer, NULL);

    pthread_join(prod, NULL);
    pthread_join(cons, NULL);

    sem_destroy(&empty);
    sem_destroy(&full);
    pthread_mutex_destroy(&mutex);
    return 0;
}
```

```bash
gcc semaphore_demo.c -o semaphore_demo -lpthread && ./semaphore_demo
```

## 常见问题

1. **`fork()` 返回 -1**：检查系统进程数限制 `ulimit -u`
2. **信号量编译报错**：需链接 pthread 库 `-lpthread`
3. **共享内存权限问题**：使用 `shm_open` 时检查 `/dev/shm` 挂载状态
